    cols = st.columns(3)

            </div>
            ''', unsafe_allow_html=True)
            if st.button(f"Go {dest['title'].split()[0]}", key=f"land_{dest['loc']}", use_container_width=True):
                enter_game(dest['loc'])
                st.rerun()
